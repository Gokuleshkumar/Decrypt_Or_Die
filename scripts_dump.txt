        async function fetchFormQr() {
            try {
                const res = await fetch('api.php?action=get_qr');
                const data = await res.json();

                if (data.success && data.paths && data.paths.length > 0) {
                    // Shuffle the paths array for random distribution
                    const paths = data.paths.sort(() => 0.5 - Math.random());
                    let currentIdx = 0;
                    const img = document.getElementById('formQrCode');
                    const container = document.getElementById('formQrContainer');
                    const unavailable = document.getElementById('formQrUnavailable');

                    function tryLoadImage() {
                        if (currentIdx >= paths.length) {
                            // All images failed
                            container.classList.add('hidden');
                            unavailable.classList.remove('hidden');
                            return;
                        }

                        img.onload = () => {
                            img.onload = null;
                            img.onerror = null;
                            container.classList.remove('hidden');
                            unavailable.classList.add('hidden');
                        };

                        img.onerror = () => {
                            currentIdx++;
                            tryLoadImage();
                        };

                        img.src = paths[currentIdx];
                    }

                    tryLoadImage();

                } else {
                    document.getElementById('formQrContainer').classList.add('hidden');
                    document.getElementById('formQrUnavailable').classList.remove('hidden');
                }
            } catch (err) {
                console.error("Failed to load QR code:", err);
                document.getElementById('formQrContainer').classList.add('hidden');
                document.getElementById('formQrUnavailable').classList.remove('hidden');
            }
        }

        const colorsArray = new Float32Array(count * 3);

        for (let i = 0; i < count * 3; i += 3) {
            // Spherical distribution
            const r = 40 * Math.cbrt(Math.random());
            const theta = Math.random() * 2 * Math.PI;
            const phi = Math.acos(2 * Math.random() - 1);

            posArray[i] = r * Math.sin(phi) * Math.cos(theta);
            posArray[i + 1] = r * Math.sin(phi) * Math.sin(theta);
            posArray[i + 2] = r * Math.cos(phi);

            // Colors: Mix of Purple, Cyan, Pink
            const colorMix = Math.random();
            if (colorMix < 0.33) { // Purple
                colorsArray[i] = 0.54; colorsArray[i + 1] = 0.36; colorsArray[i + 2] = 0.96;
            } else if (colorMix < 0.66) { // Cyan
                colorsArray[i] = 0.02; colorsArray[i + 1] = 0.71; colorsArray[i + 2] = 0.83;

                colorsArray[i] = 0.92; colorsArray[i + 1] = 0.28; colorsArray[i + 2] = 0.6;
            }

        geometry.setAttribute('color', new THREE.BufferAttribute(colorsArray, 3));

        function closeProjectRegModal() {
            const overlay = document.getElementById('project-reg-overlay');
            const content = document.getElementById('project-reg-content');
            overlay.style.opacity = '0';
            content.style.transform = 'scale(0.95)';
            setTimeout(() => {
                overlay.style.visibility = 'hidden';
            }, 300);
            document.body.style.overflow = 'auto';
        }

        async function fetchProjectQr() {
            try {
                const res = await fetch('api.php?action=get_project_qr');
                const data = await res.json();
                const img = document.getElementById('projectQrCode');
                const container = document.getElementById('projectQrContainer');
                const unavailable = document.getElementById('projectQrUnavailable');

                if (data.success && data.paths && data.paths.length > 0) {
                    const paths = data.paths.sort(() => 0.5 - Math.random());
                    img.onload = () => { container.classList.remove('hidden'); unavailable.classList.add('hidden'); };
                    img.onerror = () => { container.classList.add('hidden'); unavailable.classList.remove('hidden'); };
                    img.src = paths[0];
                } else {
                    container.classList.add('hidden');
                    unavailable.classList.remove('hidden');
                }
            } catch (err) {
                document.getElementById('projectQrContainer').classList.add('hidden');
                document.getElementById('projectQrUnavailable').classList.remove('hidden');
            }
        }

        document.getElementById('projectRegForm').addEventListener('submit', async (e) => {
            e.preventDefault();
            const form = e.target;
            const formData = new FormData(form);
            formData.append('action', 'register_project');

            const status = document.getElementById('projectRegStatus');
            const btn = document.getElementById('btnProjectSubmit');
            status.innerText = "Submitting...";
            status.classList.remove('hidden', 'text-green-500', 'text-red-500');
            status.classList.add('text-yellow-500');
            btn.disabled = true;

            try {
                const res = await fetch('api.php', { method: 'POST', body: formData });
                const data = await res.json();
                if (data.success) {
                    status.innerText = "Registration submitted successfully!";
                    status.classList.replace('text-yellow-500', 'text-green-500');
                    form.reset();
                    setTimeout(() => {
                        closeProjectRegModal();
                        // Show success modal
                        const modal = document.getElementById('successModal');
                        const modalContent = document.getElementById('successModalContent');
                        const modalMessage = document.getElementById('successModalMessage');
                        if (modalMessage) {
                            modalMessage.innerText = "Your project registration has been received! Our coordinators will verify your payment and update the status soon.";
                        }
                        if (modal && modalContent) {
                            modal.classList.remove('hidden');
                            setTimeout(() => { modalContent.style.transform = 'scale(1)'; modalContent.style.opacity = '1'; }, 50);
                        }
                    }, 1500);
                } else {
                    status.innerText = data.message || "Failed.";
                    status.classList.replace('text-yellow-500', 'text-red-500');
                }
            } catch (err) {
                status.innerText = "Server Error.";
                status.classList.replace('text-yellow-500', 'text-red-500');
            } finally {
                btn.disabled = false;
            }
        });
    </script>
